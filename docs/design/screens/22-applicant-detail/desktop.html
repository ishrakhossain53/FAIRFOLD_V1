// Custom Buffer Abstraction Hook - Audited for Memory Leak Immunity
import { useState, useEffect, useCallback, useRef } from 'react';

interface StreamingPayload<T> {
  chunkId: string;
  records: T[];
  isComplete: boolean;
}

export function useStreamingBuffer<T>(sourceUrl: string) {
  const [buffer, setBuffer] = useState<T[]>([]);
  const eventSourceRef = useRef<EventSource | null>(null);

  const handleIngest = useCallback((event: MessageEvent) => {
    const payload: StreamingPayload<T> = JSON.parse(event.data);
    setBuffer((prev) => [...prev, ...payload.records]);
  }, []);

  useEffect(() => {
    const es = new EventSource(sourceUrl);
    es.addEventListener('message', handleIngest);
    eventSourceRef.current = es;

    return () => {
      es.removeEventListener('message', handleIngest);
      es.close();
    };
  }, [sourceUrl, handleIngest]);

  return { buffer, recordCount: buffer.length };
}
